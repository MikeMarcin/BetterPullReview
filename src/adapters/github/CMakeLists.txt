if(BPR_BUILD_QT)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network)
    find_package(Threads REQUIRED)

    add_library(
        better_pull_review_adapter_github
        STATIC
            src/gh_cli_transport.cpp
            src/github_capability_probe.cpp
            src/github_cli_process_runner.cpp
            src/github_host_profile.cpp
            src/github_paged_read.cpp
            src/github_provider.cpp
            src/github_read_contract.cpp
            src/github_read_normalizer.cpp
            src/github_wire_request_validation.cpp
            src/native_https_transport.cpp
    )
    add_library(
        BetterPullReview::AdapterGitHub
        ALIAS
        better_pull_review_adapter_github
    )

    target_link_libraries(
        better_pull_review_adapter_github
        PUBLIC
            BetterPullReview::Application
        PRIVATE
            Qt6::Core
            Qt6::Network
            Threads::Threads
    )
    target_include_directories(
        better_pull_review_adapter_github
        PRIVATE
            "${CMAKE_CURRENT_SOURCE_DIR}/src"
    )
    target_compile_features(
        better_pull_review_adapter_github
        PUBLIC
            cxx_std_23
    )
    if(BPR_BUILD_TESTS)
        target_compile_definitions(
            better_pull_review_adapter_github
            PRIVATE
                BPR_NATIVE_HTTPS_TESTING=1
        )
    endif()
    if(APPLE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        target_compile_definitions(
            better_pull_review_adapter_github
            PUBLIC
                _LIBCPP_ENABLE_EXPERIMENTAL
        )
    endif()

    if(APPLE)
        find_library(
            BPR_MACOS_SECURITY_FRAMEWORK
            Security
            REQUIRED
        )
        find_library(
            BPR_MACOS_COREFOUNDATION_FRAMEWORK
            CoreFoundation
            REQUIRED
        )
        target_sources(
            better_pull_review_adapter_github
            PRIVATE
                src/macos_keychain.cpp
        )
        target_link_libraries(
            better_pull_review_adapter_github
            PRIVATE
                "${BPR_MACOS_SECURITY_FRAMEWORK}"
                "${BPR_MACOS_COREFOUNDATION_FRAMEWORK}"
        )
    endif()

    if(WIN32)
        target_sources(
            better_pull_review_adapter_github
            PRIVATE
                src/windows_credential_manager.cpp
        )
        target_link_libraries(
            better_pull_review_adapter_github
            PRIVATE
                Advapi32
        )
    endif()

    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(
            BPR_LIBSECRET
            REQUIRED
            IMPORTED_TARGET
            libsecret-1>=0.20.0
        )
        target_sources(
            better_pull_review_adapter_github
            PRIVATE
                src/linux_secret_service.cpp
        )
        target_link_libraries(
            better_pull_review_adapter_github
            PRIVATE
                PkgConfig::BPR_LIBSECRET
        )
    endif()

    bpr_enable_hardening(better_pull_review_adapter_github)
    bpr_enable_warnings(better_pull_review_adapter_github)
else()
    add_library(better_pull_review_adapter_github INTERFACE)
    add_library(
        BetterPullReview::AdapterGitHub
        ALIAS
        better_pull_review_adapter_github
    )
    target_link_libraries(
        better_pull_review_adapter_github
        INTERFACE
            BetterPullReview::Application
    )
endif()
