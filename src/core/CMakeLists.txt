add_library(BetterPullReview::Core ALIAS better_pull_review_core)

target_include_directories(
    better_pull_review_core
    PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/include"
)
target_compile_features(better_pull_review_core PUBLIC cxx_std_23)
# Approved dependency inventory; the boundary check enforces exact links.
# Semantic eligibility follows the architecture external-determinism contract.
target_link_libraries(better_pull_review_core PRIVATE BetterPullReview::Xdiff BetterPullReview::RE2 BetterPullReview::ZLib BetterPullReview::UTF8Proc BetterPullReview::Sha1dc BetterPullReview::HaclSha2 BetterPullReview::RapidFuzz BetterPullReview::Dtl)

# Reproducible portable builds may explicitly keep the approved HACL backend.
option(BPR_CORE_SHA256_PORTABLE "Use portable HACL SHA-256 instead of platform crypto" OFF)
if(BPR_CORE_SHA256_PORTABLE)
    target_sources(better_pull_review_core PRIVATE src/sha256_portable.cpp)
elseif(APPLE)
    target_sources(better_pull_review_core PRIVATE src/sha256_commoncrypto.cpp)
elseif(WIN32)
    target_sources(better_pull_review_core PRIVATE src/sha256_cng.cpp)
    target_link_libraries(better_pull_review_core PRIVATE bcrypt)
else()
    target_sources(better_pull_review_core PRIVATE src/sha256_portable.cpp)
endif()

bpr_enable_hardening(better_pull_review_core)
bpr_enable_warnings(better_pull_review_core)
